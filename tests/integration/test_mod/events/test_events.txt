# Test events for QuickJS integration testing.
#
# These events exercise the effect and trigger hooks.

# Main integration test event
country_event = {
	id = test_quickjs.1
	title = l_TEST_EVENT_TITLE
	desc = l_TEST_EVENT_DESC
	
	# Trigger: Only fire for player country
	trigger = {
		is_country_type = default
		test_custom_trigger = yes
	}
	
	# Fire on game start for testing
	fire_only_once = yes
	
	immediate = {
		log = "test_quickjs.1: Integration test event fired"
		set_country_flag = test_quickjs_flag
		
		# Execute test effects
		test_custom_effect = yes
		test_logging_effect = yes
	}
	
	option = {
		name = "Excellent"
		log = "test_quickjs.1: Player acknowledged test event"
	}
}
